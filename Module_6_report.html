Results & Analysis

The Mandelbrot Set is defined as the set of all complex numbers $c$ such that $|0|,|f_c(0)|,|f_c(f_c(0))|, \cdots$ is bounded, where $f_c(z) := z^2 +c$. In other words, repeated iterations of the function keep the origin bounded. To visualize this, we first need to create a grid and plot some points to see where they land after we apply the transformation. 


	import numpy as np
import matplotlib.pyplot as plt

# Create a simple 2D grid of points using NumPy
def generate_grid(size):
    x = np.linspace(-3, 3, size)  # Generate 'size' number of points between -3 and 3, this can vary
    y = np.linspace(-3, 3, size)
    grid = np.meshgrid(x, y)  # Create a 2D grid of points
    return grid

# Visualize the grid of points
def plot_grid(grid):
    x_grid, y_grid = grid
    plt.figure(figsize=(10, 10))
    plt.scatter(x_grid, y_grid, s=1, color='blue')  # Plot the grid points
    plt.title(f'2D Grid of {len(x_grid)}x{len(y_grid)} points')
    plt.xlim(-3, 3) # Keep the axes fixed
    plt.ylim(-3, 3)
    plt.gca().set_aspect('equal') # Maintain aspect ratio
    plt.show()


# Set the size of the grid
size = 50  # You can adjust this to make the grid denser or sparser

# Generate and plot the grid
grid = generate_grid(size)
plot_grid(grid)

Then, we just need to define the transformation that we want to apply and specify the parameter $c$. 


	def iterative_transform(grid, iterations, c):
    x_grid, y_grid = grid
    z = x_grid + 1j * y_grid  # Create a complex plane
    for i in range(iterations):
        z = z ** 2 + c  # Iteratively apply the transformation
    return np.real(z), np.imag(z)

# Set the number of iterations and parameter
iterations = 1
c= -0.5

for i in range(0, 11):
  # Apply the iterative transformation
  fractal_grid = iterative_transform(grid, i, c)

  # Plot the fractal grid
  plot_grid(fractal_grid)
